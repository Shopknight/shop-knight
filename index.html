function showMessage(text,ok){   msg.textContent=text;   msg.className="msg "+(ok?"ok":"err"); } form.addEventListener("submit",async e=>{   e.preventDefault();   btn.disabled=true;   btn.textContent="در حال ثبت سفارش...";   try{     const coins=document.getElementById("coins");     const selected=coins.options[coins.selectedIndex];     const file=document.getElementById("receipt").files[0];     if(!file) throw new Error("لطفاً تصویر رسید را انتخاب کنید.");     if(!file.type.startsWith("image/")) throw new Error("فایل رسید باید تصویر باشد.");     if(file.size>10*1024*1024) throw new Error("حجم تصویر نباید بیشتر از ۱۰ مگابایت باشد.");     const orderId=crypto.randomUUID();     const ext=(file.name.split(".").pop()"jpg").toLowerCase().replace(/[^a-z0-9]/g,"")"jpg";     const receiptPath=orders/${orderId}.${ext};     let response=await fetch(       ${SUPABASE_URL}/storage/v1/object/receipts/${receiptPath},       {         method:"POST",         headers:{           apikey:SUPABASE_KEY,           Authorization:Bearer ${SUPABASE_KEY},           "Content-Type":file.type,           "x-upsert":"false"         },         body:file       }     );     if(!response.ok){       const t=await response.text();       throw new Error("آپلود رسید انجام نشد.\n"+t);     }     const orderData={       id:orderId,       player_id:document.getElementById("player_id").value.trim(),       coins:Number(selected.value),       price:Number(selected.dataset.price),       name:document.getElementById("name").value.trim()||null,       contact:document.getElementById("contact").value.trim()||null,       status:"در انتظار",       receipt_path:receiptPath     };     response=await fetch(${SUPABASE_URL}/rest/v1/orders,{       method:"POST",       headers:{         apikey:SUPABASE_KEY,         Authorization:Bearer ${SUPABASE_KEY},         "Content-Type":"application/json",         Prefer:"return=minimal"       },       body:JSON.stringify(orderData)     });     if(!response.ok){       const t=await response.text();       throw new Error("ثبت سفارش در پایگاه‌داده انجام نشد.\n"+t);     }     showMessage("✅ سفارش با موفقیت ثبت شد.\nکد سفارش: "+orderId+"\nرسید پرداخت هم ارسال شد.",true);     form.reset();   }catch(err){     console.error(err);     showMessage("❌ "+(err.message||"خطای نامشخص"),false);   }finally{     btn.disabled=false;     btn.textContent="ثبت سفارش 👑";   } });
